\begin{equation}
h_1\sim_{\rm pred}h_2
\quad\Longleftrightarrow\quad
\begin{cases}
\mathcal L(h_1)=\mathcal L(h_2),\\
\mathcal O_{h_1}(w)=\mathcal O_{h_2}(w)\quad\forall w\in\mathcal L(h_1).
\end{cases}
\label{eq:predictive}
\end{equation}
