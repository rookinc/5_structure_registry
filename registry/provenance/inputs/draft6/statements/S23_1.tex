\setcounter{theorem}{0}
\begin{proposition}
If the address graph has finite diameter $D$, then for every sequence of histories with event depths tending to infinity,
\[
0\leq\frac{d_{\mathrm{field}}(s(\gamma_n),t(\gamma_n))}{|\gamma_n|_{\mathrm{event}}}
\leq\frac{D}{|\gamma_n|_{\mathrm{event}}}\longrightarrow0.
\]
\end{proposition}
