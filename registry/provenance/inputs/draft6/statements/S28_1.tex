\setcounter{theorem}{0}
\begin{proposition}[Undisturbed arbitrary payloads have no state-dependent pure receipt]
Suppose a common isometry obeys
\[
V|\psi\rangle=U|\psi\rangle\otimes|r_\psi\rangle
\]
for every normalized input, where $U$ is fixed and isometric. Then the receipt cannot distinguish arbitrary input states.
\end{proposition}
\begin{proof}
Inner-product preservation gives
$\langle\phi|\psi\rangle=\langle\phi|\psi\rangle\langle r_\phi|r_\psi\rangle$.
For nonorthogonal inputs, $\langle r_\phi|r_\psi\rangle=1$. Orthogonal states can each be linked to a common nonorthogonal superposition, giving the same conclusion throughout the input space.
\end{proof}
